// ============================================================ // PROCESSAR LISTA DE PRESENÇA - VERSÃO ULTRA SIMPLES // ============================================================ function processPresence() { console.log('🔄 processPresence() chamada!'); const text = document.getElementById('presenceList').value; console.log('📝 Texto recebido (primeiros 200 caracteres):', text.substring(0, 200)); if (!text || !text.trim()) { showToast('⚠️ Por favor, cole uma lista de presença!'); return; } // Divide o texto em linhas const lines = text.split('\n'); console.log(`📄 Total de linhas: ${lines.length}`); let encontrados = 0; const novosJogadores = []; const jogadoresAtualizados = []; // Para CADA linha do texto for (let i = 0; i < lines.length; i++) { const linha = lines[i].trim(); if (!linha) continue; // Procura por: NÚMERO + NOME + ✅ ou 👍 // Exemplos: "1- DAN✅", "3- MARCOS✅", "15- HELTON👍" const regex = /^(\d+)\s*[-.)]?\s*([A-Za-záéíóúãõâêîôûàèìòùç\s\.]+?)\s*([✅👍])/; const match = linha.match(regex); if (match) { const numero = match[1]; let nome = match[2].trim(); const emoji = match[3]; // Limpa o nome: remove caracteres especiais, mantém letras, espaços, pontos nome = nome.replace(/[^\w\sáéíóúãõâêîôûàèìòùç\-\.]/gi, '').trim(); console.log(`✅ Linha ${i+1}: Número ${numero} - Nome: "${nome}" - Emoji: ${emoji}`); if (nome && nome.length >= 2) { // Verifica se já existe const existe = players.find(p => p.name.toLowerCase() === nome.toLowerCase()); if (existe) { existe.confirmed = true; jogadoresAtualizados.push(nome); console.log(`♻️ Jogador atualizado: ${nome}`); } else { const nivelPadrao = parseInt(document.getElementById('defaultLevel').value) || 3; const posicao = autoAssignPosition(nome); const genero = Math.random() > 0.5 ? 'M' : 'F'; novosJogadores.push({ name: nome, gender: genero, position: posicao, level: nivelPadrao, confirmed: true }); console.log(`🆕 Novo jogador: ${nome} - ${posicao} - ⭐${nivelPadrao}`); } encontrados++; } } else { // Se não encontrou com a regex, tenta um método mais simples // Verifica se a linha tem ✅ ou 👍 e tenta extrair o nome if (linha.includes('✅') || linha.includes('👍')) { let nome = linha; // Remove o número no início nome = nome.replace(/^\d+\s*[-.)]?\s*/, ''); // Remove emojis nome = nome.replace(/[✅👍❌]/g, ''); // Remove caracteres especiais nome = nome.replace(/[^\w\sáéíóúãõâêîôûàèìòùç\-\.]/gi, ''); nome = nome.trim(); if (nome && nome.length >= 2) { console.log(`📝 Linha ${i+1} (alternativo): "${nome}"`); const existe = players.find(p => p.name.toLowerCase() === nome.toLowerCase()); if (!existe) { const nivelPadrao = parseInt(document.getElementById('defaultLevel').value) || 3; const posicao = autoAssignPosition(nome); const genero = Math.random() > 0.5 ? 'M' : 'F'; novosJogadores.push({ name: nome, gender: genero, position: posicao, level: nivelPadrao, confirmed: true }); console.log(`🆕 Novo jogador (alternativo): ${nome}`); encontrados++; } } } } } console.log(`📊 Total de jogadores encontrados: ${encontrados}`); console.log(`🆕 Novos: ${novosJogadores.length}`); console.log(`♻️ Atualizados: ${jogadoresAtualizados.length}`); // Adiciona os novos jogadores if (novosJogadores.length > 0) { players = [...players, ...novosJogadores]; console.log(`✅ ${novosJogadores.length} novos jogadores adicionados`); } // Salva e atualiza savePlayersToServer(); updateUI(); const confirmados = players.filter(p => p.confirmed).length; let msg = `✅ Processado! ${encontrados} jogadores confirmados encontrados.`; msg += `\nTotal: ${confirmados} jogadores.`; if (novosJogadores.length > 0) msg += `\n🆕 Novos: ${novosJogadores.length}`; if (jogadoresAtualizados.length > 0) msg += `\n♻️ Atualizados: ${jogadoresAtualizados.length}`; console.log(`📊 ${msg}`); showToast(msg); document.getElementById('presenceList').value = ''; }
